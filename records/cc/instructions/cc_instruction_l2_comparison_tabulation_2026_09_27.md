```python
def fence_ranges(lines):
    """1-based (first, last) of every fenced code block, fence lines included; an unclosed fence
    runs to the end of the file.  The fence test is FENCE, the one `headings_by_line` uses."""
    out = []
    start = None
    for number, line in enumerate(lines, 1):
        if FENCE.match(line):
            if start is None:
                start = number
            else:
                out.append((start, number))
                start = None
    if start is not None:
        out.append((start, len(lines)))
    return out


def block_around(lines, number, fences):
    """THE PASSAGE RULE (THE_PASSAGE_RULE, (a) to (c)) for one 1-based line."""
    for first, last in fences:
        if first <= number <= last:
            return first, last
    fenced = set()
    for first, last in fences:
        fenced.update(range(first, last + 1))

    def stops(n):
        return lines[n - 1].strip() == "" or n in fenced

    first = last = number
    while first > 1 and not stops(first - 1):
        first -= 1
    while last < len(lines) and not stops(last + 1):
        last += 1
    if first == last and HEADING.match(lines[first - 1]):
        following = last + 1
        while following <= len(lines) and lines[following - 1].strip() == "":
            following += 1
        if (following <= len(lines) and following not in fenced
                and not HEADING.match(lines[following - 1])):
            last = block_around(lines, following, fences)[1]
    return first, last


def merge_ranges(ranges):
    """The sorted union of 1-based inclusive ranges; overlapping or touching ranges merge."""
    out = []
    for first, last in sorted(ranges):
        if out and first <= out[-1][1] + 1:
            out[-1] = (out[-1][0], max(out[-1][1], last))
        else:
            out.append((first, last))
    return out


def subtract_spans(ranges, spans):
    """`ranges` with every line inside any (first, last) of `spans` removed, a range split where a
    span cuts it; the result merged."""
    out = []
    for first, last in ranges:
        pieces = [(first, last)]
        for s_first, s_last in spans:
            kept = []
            for a, b in pieces:
                if b < s_first or a > s_last:
                    kept.append((a, b))
                    continue
                if a < s_first:
                    kept.append((a, s_first - 1))
                if b > s_last:
                    kept.append((s_last + 1, b))
            pieces = kept
        out.extend(pieces)
    return merge_ranges(out)
```

## Task 2 — the tabulation — the reading file, ONE commit PER MEMBER, in the published order

**2(a) — the unit, fixed before you begin.** An **OUTGOING STATEMENT** is one sentence, one bullet or
one table row of a member's text that states what the analysis does, must do, may assume, or must not do
— about L2's subject (the charter's question: the tonality at each moment, where one harmony gives way
to the next, which sounding notes belong to the harmony and which elaborate it, and what chord is read
over each span; what L2 publishes: the segmentation, the tonality per span, the chord as degree,
quality, figure and applied target, the chord-tone assignment and elaboration relation, the rivals with
their mass; and the faces (a) to (h) as the brief's §2 and the derivation's §3 headings name them), or
about another charter's subject met inside the member (placed RELOCATED). A sentence that is none of
those — narrative, provenance, a test count, a commit hash, a pointer to another document — is NOT an
outgoing statement and is listed once per member under *not a statement*, so that completeness can be
checked by arithmetic at the member. **A paragraph is never a unit.** Where one sentence carries two
claims, it is two statements, each quoted with the sentence and marked (i) and (ii). **For a passage
member, tabulate only the lines inside its published ranges**; a sentence that starts before a range or
runs past it is quoted whole and tabulated once, at the member holding its first line inside a range.

**2(b) — per outgoing statement**, a row numbered **Row M.N** (M the member's position, N the statement's
order in it): quote it verbatim with its location (document and nearest heading; a line number only as a
locator beside the heading, never as the anchor); name the derived statement or statements (by
L2-S-number) that speak to it, or state that none does; on the **CURRENT-TEXT AXIS** say, for each named
derived statement, exactly one of **AGREES / DIFFERS / THE DERIVATION IS SILENT** — evidence about the
present text, never a verdict on the derivation; then the **PROPOSED DISPOSITION**, exactly one of:

- **ADOPTED — carried**: a derived statement carries its content in substance (name it).
- **ADOPTED — proposed**: no derived statement carries it, it is not implementation-describing, not
  another charter's, not historical, and the worth test does not discard it. **A PROPOSAL for the
  user: the outgoing content would be added to the derived specification.** One sentence beside the
  row. Never write it into the derivation.
- **RELOCATED**: its content belongs to another charter. Name the charter by `FRAMEWORK.md` §5's own
  words — *L0 — The notated record* (the input contract), *L1 — Change points, candidates and notated
  evidence*, *L3 — The read-off facts*, *the second axis — voice leading*, or one of the NOT A LAYER
  items *the uncertainty surface* or *the measurement of the analysis*. Where the input contract
  already carries the content, name its statement (IC S-n). Append the row to the TRANSFER LIST section.
- **QUARANTINED**: it states how the implementation currently works, or a measured property of it.
  Write the audit question in one sentence beside the row.
- **DISCARDED**: under principle #10's worth test, with the finding, the date the row is written, and
  the reason — all three, or the disposition does not reach the row (D-677's shape).
- **HISTORICAL**: an event, a build state, a superseded plan, a status.
- **UNPLACED**: you cannot defend one disposition in one sentence at the two texts. Say what you read.
